Click anywhere on the page to enable sound."); }); } // Set initial word tick.value = prepareWord(rawWords[0]); let i = 0; setInterval(function() { i = (i + 1) % rawWords.length; tick.value = prepareWord(rawWords[i]); }, 3000); }
